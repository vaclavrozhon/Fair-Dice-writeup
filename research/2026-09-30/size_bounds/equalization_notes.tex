\begin{proposition}[An improved known relation between the optima]
Let $A_n$ be the shortest arbitrary-composition permutation-fair string
length, and let $E_n$ be the corresponding equal-composition optimum.
The existing Hahn construction and exponential lower bound already imply
\[
 A_n\le E_n\le A_n^{O(\log n)}
                 \le A_n^{O(\log\log A_n)}.
\]
This is stronger than the relation $E_n\le A_n^{O(\log A_n)}$
obtained by independent coordinate blowups alone.
\end{proposition}
\begin{proof}
For absolute positive constants $c,C$ and all sufficiently large $n$,
$\log A_n\ge cn$ and $\log E_n\le Cn\log n$. Consequently
$\log E_n\le(C/c)\log A_n\log n$. Finally $n=O(\log A_n)$.
\end{proof}

\begin{proposition}[A sharper independent-blowup bound]
For an $n$-fair string of total length $A$ and counts $c_1,\ldots,c_n$,
one can equalize to length
\[
 n\operatorname{lcm}(c_1,\ldots,c_n)
 \le\frac{n\prod_i c_i}{\prod_{p\le n}p^{n-p}}
 \le\frac{n(A/n)^n}{\prod_{p\le n}p^{n-p}}.
\]
\end{proposition}
\begin{proof}
Blow up letter $i$ by $\operatorname{lcm}(c_1,\ldots,c_n)/c_i$.
At least $n-p+1$ counts are divisible by each prime $p\le n$.
Therefore
\[
 v_p\left(\frac{\prod_i c_i}{\operatorname{lcm}(c_1,\ldots,c_n)}\right)
 =\sum_i v_p(c_i)-\max_i v_p(c_i)\ge n-p.
\]
Multiply these estimates and apply AM--GM.
\end{proof}

\begin{lemma}[Concatenation with proportional compositions]
Let $s_1,\ldots,s_B$ be permutation-fair strings on the same alphabet,
and suppose their multiplicity vectors are $a_b c$ for a fixed vector
$c\in\mathbb R_{>0}^n$ and positive scalars $a_b$.
Then $s_1\cdots s_B$ is permutation-fair, with vector
$(\sum_b a_b)c$.
\end{lemma}
\begin{proof}
Fix a pattern $\pi$ on a set $S$ of distinct letters and let $q_b$ be
the number of its consecutive letters selected from block $b$.
Partial fairness gives contribution
\[
 \left(\prod_{i\in S}c_i\right)
 \prod_{b=1}^B\frac{a_b^{q_b}}{q_b!}.
\]
Summing over $q_1+\cdots+q_B=|S|$ gives
\[
 C_\pi(s_1\cdots s_B)=
 \frac{(\sum_b a_b)^{|S|}}{|S|!}\prod_{i\in S}c_i,
\]
independently of the order of $\pi$.
\end{proof}
