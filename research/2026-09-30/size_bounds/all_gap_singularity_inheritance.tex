Let $G_A$ be the reduced free nilpotent group on an alphabet $A$, and let
$X_i$ denote its degree-one generators. For a positive word $W$, put
\[
 S(W)=\operatorname{span}\{\operatorname{Ad}_{P_q}X_i:
                i\in A,\ q\text{ a gap of }W\}\subseteq\mathfrak g_A,
\]
where $P_q$ is the signature of the prefix ending at the gap. This is the
right-trivialized differential of inserting arbitrary infinitesimal
letter weights at all gaps. Write $H=\sum_{i\in A}X_i$.

\begin{lemma}
If $W$ is fair with signature $\exp(MH)$, then
\[
 S(W^t)=\sum_{j=0}^{t-1}e^{jM\operatorname{ad}H}S(W).
\]
Consequently, if $S(W)$ is invariant under $\operatorname{ad}H$, then
$S(W^t)=S(W)$ for every positive integer $t$.
\end{lemma}
\begin{proof}
Every prefix of $W^t$ is the signature of some number $j$ of complete
copies, followed by a prefix $P$ of one copy. Its adjoint action is
$\operatorname{Ad}_{\exp(jMH)P}=e^{jM\operatorname{ad}H}\operatorname{Ad}_P$.
The gap at the final endpoint contributes no new vectors, since it is
already a gap of the last copy. Invariance gives the last assertion;
the first copy supplies the reverse inclusion.
\end{proof}

\begin{lemma}
Let $V$ be a word on $B\supseteq A$ whose restriction to $A$ is $W^t$.
Under the natural Lie-algebra projection $\pi:\mathfrak g_B\to
\mathfrak g_A$, one has
\[
 \pi(S(V))\subseteq S(W^t).
\]
In particular, if $S(W^t)\ne\mathfrak g_A$, then
$S(V)\ne\mathfrak g_B$.
\end{lemma}
\begin{proof}
Projection commutes with the adjoint action. An insertion of a letter
outside $A$ projects to zero. An insertion of a letter of $A$ projects
to its insertion at the corresponding gap of the restricted word.
A surjective ambient tangent space would have surjective projection.
\end{proof}

\begin{corollary}
There are positive fair words with singular all-gap endpoint maps on
arbitrarily large alphabets.
\end{corollary}
\begin{proof}
An exact rational-arithmetic diagnostic supplies one fair word $W$ with
$\dim S(W)=86<89=\dim\mathfrak g_A$ and
$[H,S(W)]\subseteq S(W)$. The files
\texttt{certify\_all\_gap\_singularity.py} and
\texttt{certify\_all\_gap\_power\_singularity.py} verify these assertions
by integer matrix factorizations through an invertible $86\times86$
minor, independently of floating-point arithmetic. The input is the
published fair word stored in
\texttt{../equal\_size/meyer\_2023\_five\_dice\_sixty.json}.
Apply the standard one-letter insertion construction repeatedly. At each
stage, restriction to the previous alphabet is a positive integer power
of the preceding word. Hence restriction to $A$ remains a power of $W$.
The two lemmas give singularity at every stage.
\end{proof}

\begin{corollary}[Fairness does not force truncated surjectivity]
For every $r\ge5$ and every $n\ge2r$, there is a fully
permutation-fair positive word on $n$ letters whose all-gap endpoint
map is not surjective even onto the reduced Lie algebra truncated
through degree $r$. In particular, fairness through degree $2r$
does not imply full endpoint regularity through degree $r$.
\end{corollary}
\begin{proof}
Use the same inherited construction on $n$ letters. Projection to
its original five-letter alphabet factors through truncation at
degree $r$, since that entire reduced Lie algebra has degree at
most five. Surjectivity of the truncated tangent map would therefore
surject onto the original five-letter algebra, contradicting the
inherited singularity. Full fairness implies fairness through $2r$.
The actual face-weight differential is a subspace of the all-gap
insertion differential, so it cannot restore this surjectivity.
\end{proof}

\noindent\textbf{Scope.} This closes the proposed route that every
positive fair word must have a regular all-gap endpoint map, even after
repetition, and also the stronger-accuracy-to-lower-degree surjectivity
variant. It gives no lower bound on the relative corank and does not
exclude a weaker factorial lower bound for the rank. The fixed input is
used solely to falsify a universal asymptotic lemma; no small-size
optimization is being pursued.
